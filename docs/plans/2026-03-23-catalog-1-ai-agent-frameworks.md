```python
def generate_catalog_readme(
    entries: list[dict],
    topic: str,
    output_dir: Path,
    template_dir: Path | None = None,
) -> None:
    """Generate a per-catalog README.md from catalog data."""
    from collections import defaultdict
    from datetime import datetime
    from jinja2 import Environment, FileSystemLoader

    if template_dir is None:
        template_dir = Path(__file__).parent / "templates"

    output_dir.mkdir(parents=True, exist_ok=True)

    domain_entries = defaultdict(list)
    for e in entries:
        for d in e.get("found_in_domains", [e.get("sub_domain", "unknown")]):
            domain_entries[d].append(e)

    domains_summary = []
    for d_name in sorted(domain_entries.keys()):
        d_entries = domain_entries[d_name]
        avg = sum(e.get("score", 0) for e in d_entries) / len(d_entries)
        domains_summary.append({"name": d_name, "count": len(d_entries), "avg_score": f"{avg:.1f}"})

    scores = [e.get("score", 0) for e in entries]
    top_repos = [e for e in entries if e.get("score", 0) >= 8]

    # Check if capabilities/ dir exists
    has_capabilities = (output_dir / "capabilities").is_dir() and any((output_dir / "capabilities").iterdir())

    context = {
        "topic": topic,
        "generated_at": datetime.now().strftime("%Y-%m-%d"),
        "total": len(entries),
        "domain_count": len(domain_entries),
        "min_score": min(scores) if scores else 0,
        "max_score": max(scores) if scores else 0,
        "domains": domains_summary,
        "top_repos": top_repos,
        "has_capabilities": has_capabilities,
    }

    env = Environment(loader=FileSystemLoader(str(template_dir)))
    tpl = env.get_template("catalog-readme.md.jinja")
    readme_path = output_dir / "README.md"
    readme_path.write_text(tpl.render(**context))
    print(f"[catalog-readme] Wrote {readme_path}")
```
